# 알파Plus 실물 재현 구현 계획

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 알파Plus 실물 보통약관(48조·7관·정적 표·박스) + 담보 5건을 시스템 데이터로 모델링해 조립 결과가 원문과 같아지게 하고, 그 과정을 Vitest 대조 + Playwright 하이브리드 E2E 로 고정한다.

**Architecture:** 문면 노드 트리에 `section`·`table`·`box` 3종을 더하고(허용 자식 규칙 확장, ADR-0029), 별표 번호를 상품 별표 목록 순서로 바꾼다(ADR-0030). 실물 문면은 파싱양식 `.md` → 노드 트리 변환 스크립트(`scripts/terms/`)가 만들어 시드 JSON 정본에 넣고, 대조기(`src/domain/assembly/compare.ts`)가 조립 결과를 파싱양식 텍스트로 되돌려 원문과 조 단위로 비교한다.

**Tech Stack:** TypeScript · Next 16 (server actions) · drizzle + PGlite · Vitest · Playwright · tsx.

**Spec:** `docs/superpowers/specs/2026-09-07-알파플러스-실물재현-design.md`

## Global Constraints

- `src/domain/**` 은 순수층 — DB·React import 금지 (ESLint 강제).
- 노드 id 는 트리 안에서 유일. 번호는 저장하지 않는다 (계산값).
- 커밋 메시지에 Co-Authored-By 등 서명·트레일러 없음. 아이템별 커밋, 본문에 왜를 남긴다.
- `npm test` · `npm run typecheck` · `npm run lint` 녹색이 커밋 조건.
- `package.json` 의 `overrides.typescript` 유지.
- 실물 텍스트는 시드 정본(`src/db/seed/data/*.json`)과 `tests/fixtures/terms/*.md` 로 커밋한다 (2026-09-07 유저 결정).
- 스펙 수정 (계획 작성 중 확정): 공용조항은 이번 데이터에 쓰지 않는다 (소멸 조 문구가 담보마다 달라 옵션으로 못 묶고, 특약 조연결도 하지 않는다 — 실물에는 준용 문장이 없으므로 조연결하면 없는 문장이 생긴다). 표·박스 자리는 `article.children` · `paragraph.items` 두 곳 (`item.subitems` 는 실물에 없어 제외). 상품 레벨 구분자는 두지 않는다 (평균공시이율 const 의 소비처가 표 셀뿐이라 제외).

`Command` 유니온에:

`applyCommand` 의 switch 에 (기존 `setText` 케이스와 같은 패턴 — 색인에서 노드를 찾아 복제 트리에서 필드를 바꾼다):
```ts
    case "setTable": {
      const e = ix.nodes.get(cmd.nodeId);
      if (!e) return notFound(`노드 ${cmd.nodeId}`);
      if (e.node.kind !== "table") return structure("표가 아닌 노드에는 표 내용을 넣을 수 없습니다", e.path);
      if (cmd.columns.length === 0) return structure("표에는 열이 하나 이상 있어야 합니다", e.path);
      const bad = cmd.rows.findIndex((row) => row.cells.length !== cmd.columns.length);
      if (bad >= 0) return structure(`${bad + 1}번째 행의 셀 수(${cmd.rows[bad].cells.length})가 열 수(${cmd.columns.length})와 다릅니다`, e.path);
      const node = e.node as TableNode;
      delete node.title;
      if (cmd.title !== undefined && cmd.title !== "") node.title = cmd.title;
      node.columns = cmd.columns.map((c) => (c.width !== undefined ? { width: c.width } : {}));
      node.rows = cmd.rows.map((r) => ({ ...(r.header ? { header: true } : {}), cells: [...r.cells] }));
      return ok(tree);
    }
    case "setBox": {
      const e = ix.nodes.get(cmd.nodeId);
      if (!e) return notFound(`노드 ${cmd.nodeId}`);
      if (e.node.kind !== "box") return structure("박스가 아닌 노드에는 박스 내용을 넣을 수 없습니다", e.path);
      (e.node as BoxNode).title = cmd.title;
      (e.node as BoxNode).lines = [...cmd.lines];
      return ok(tree);
    }
```
`setTitle` 케이스의 종류 검사(`article` · `document`)에 `section` 을 추가한다. (`applyCommand` 가 트리를 먼저 깊은 복사하고 그 위의 색인을 쓰는 기존 구조를 따른다 — `setText` 케이스를 열어 같은 방식으로.)

import type { ErrorNode, RArticle, RDoc, RSection } from "./types";

/** 관을 투명하게 펼친 조 목록 (오류 마커 제외). */
export function articlesOf<I>(doc: RDoc<I>): RArticle<I>[] {
  const out: RArticle<I>[] = [];
  for (const c of doc.children) {
    if (c.kind === "article") out.push(c);
    else if (c.kind === "section") for (const a of c.children) if (a.kind === "article") out.push(a);
  }
  return out;
}

/** 조마다 fn 을 적용해 문서를 다시 만든다. null 이면 그 조를 뺀다. 관·오류 마커는 유지. */
export function mapArticles<I, J>(doc: RDoc<I>, fn: (a: RArticle<I>) => RArticle<J> | ErrorNode | null): RDoc<J> {
  const one = (c: RArticle<I> | ErrorNode): (RArticle<J> | ErrorNode)[] => {
    if (c.kind === "error") return [c];
    const r = fn(c);
    return r === null ? [] : [r];
  };
  return {
    ...doc,
    children: doc.children.flatMap((c): (RArticle<J> | RSection<J> | ErrorNode)[] => {
      if (c.kind === "section") return [{ ...c, children: c.children.flatMap(one) }];
      return one(c);
    }),
  };
}

import type { RenderedArticle, RenderedDoc, RenderedInline, RenderedParagraph, RenderedStatic } from "./types";

const inlineText = (list: readonly RenderedInline[]): string => list.map((n) => (n.kind === "text" ? n.text : n.kind === "error" ? `⟦${n.issue.kind}⟧` : n.label)).join("");

function staticLines(n: RenderedStatic): string[] {
  if (n.kind === "box") return ["```용어풀이", `【${n.title}】`, ...n.lines, "```"];
  const out = ["```표", `제목: ${n.title ?? "(없음)"}`];
  let separated = false;
  for (const row of n.rows) {
    out.push(`|${row.cells.join("|")}|`);
    if (row.header && !separated) { out.push(`|${n.columns.map(() => "---").join("|")}|`); separated = true; }
  }
  if (!separated && n.rows.length > 0) out.splice(3, 0, `|${n.columns.map(() => "---").join("|")}|`);
  out.push("```");
  return out;
}

function articleLines(a: RenderedArticle): string[] {
  const out = [`## ${a.label}(${a.title})`];
  const paragraphs = a.children.filter((c): c is RenderedParagraph => c.kind === "paragraph");
  const single = paragraphs.length === 1;
  for (const c of a.children) {
    if (c.kind === "error") { out.push(`= ⟦${c.issue.kind}⟧`); continue; }
    if (c.kind !== "paragraph") { out.push(...staticLines(c)); continue; }
    out.push(`${single ? "=" : "@"} ${inlineText(c.children)}`);
    for (const it of c.items ?? []) {
      if (it.kind === "error") continue;
      if (it.kind !== "item") { out.push(...staticLines(it)); continue; }
      out.push(`  - ${inlineText(it.children)}`);
      for (const s of it.subitems ?? []) if (s.kind === "subitem") out.push(`    - ${inlineText(s.children)}`);
    }
  }
  return out;
}

export function renderedToLines(doc: RenderedDoc): string[] {
  const out: string[] = [];
  for (const c of doc.children) {
    if (c.kind === "error") continue;
    if (c.kind === "section") { out.push(`# ${c.label} ${c.title}`); for (const a of c.children) if (a.kind === "article") out.push(...articleLines(a)); }
    else out.push(...articleLines(c));
  }
  return out;
}

export function sourceToLines(markdown: string): string[] {
  return markdown
    .split(/\r?\n/)
    .map((l) => l.replace(/<!--.*?-->/g, "").replace(/\s+$/, ""))
    .filter((l, i) => l.trim() !== "" && !l.startsWith("> ") && !(i === 0 && l.startsWith("# ") && !/^# 제\d+관/.test(l)));
}

export function normalizeLine(line: string): string {
  return line.replace(/제\d+조(의\d+)?\(/g, "제§조(").replace(/제\d+관/g, "제§관").replace(/\s+/g, "");
}

export interface ArticleDiff { index: number; title: string; expected: string[]; actual: string[] }

function chunks(lines: string[]): { title: string; lines: string[] }[] {
  const out: { title: string; lines: string[] }[] = [];
  for (const l of lines) {
    const m = /^##\s*제[^(]*\((.*)\)\s*$/.exec(l);
    if (m) out.push({ title: m[1], lines: [] });
    else if (out.length === 0) out.push({ title: "(머리)", lines: [l] });
    else out[out.length - 1].lines.push(l);
  }
  return out;
}

export function diffByArticle(expected: string[], actual: string[]): ArticleDiff[] {
  const e = chunks(expected), a = chunks(actual);
  const n = Math.max(e.length, a.length);
  const out: ArticleDiff[] = [];
  for (let i = 0; i < n; i++) {
    const x = e[i], y = a[i];
    const same = x && y && x.title.replace(/\s+/g, "") === y.title.replace(/\s+/g, "") && x.lines.map(normalizeLine).join("\n") === y.lines.map(normalizeLine).join("\n");
    if (!same) out.push({ index: i, title: x?.title ?? y?.title ?? "", expected: x?.lines ?? [], actual: y?.lines ?? [] });
  }
  return out;
}

```ts
export function parseTerms(markdown: string): ParsedDoc {
  const lines = markdown.split(/\r?\n/);
  const doc: ParsedDoc = { title: "", sections: [] };
  let section: ParsedDoc["sections"][number] | undefined;
  let article: ParsedArticle | undefined;
  let fence: { kind: "table" | "box" | "figure"; lines: string[] } | undefined;
  const ensureSection = () => { if (!section) { section = { title: "", articles: [] }; doc.sections.push(section); } return section; };
  const push = (l: Line) => { if (!article) throw new Error(`조 밖의 본문: ${JSON.stringify(l)}`); article.body.push(l); };
  for (const raw of lines) {
    const line = raw.replace(/<!--.*?-->/g, "").replace(/\s+$/, "");
    if (fence) {
      if (line === "```") { push(fenceToLine(fence)); fence = undefined; } else fence.lines.push(line);
      continue;
    }
    if (line === "") continue;
    if (line.startsWith("```표")) { fence = { kind: "table", lines: [] }; continue; }
    if (line.startsWith("```용어풀이")) { fence = { kind: "box", lines: [] }; continue; }
    if (line.startsWith("```그림")) { fence = { kind: "figure", lines: [] }; continue; }
    if (line.startsWith("> ")) continue;
    let m: RegExpExecArray | null;
    if ((m = /^# 제(\d+)관\s+(.*)$/.exec(line))) { section = { title: m[2].trim(), articles: [] }; doc.sections.push(section); article = undefined; continue; }
    if (line.startsWith("# ")) { if (!doc.title) doc.title = line.slice(2).trim(); continue; }
    if ((m = /^## 제(\d+)조(?:의(\d+))?\((.*)\)\s*$/.exec(line))) {
      article = { number: m[2] ? `${m[1]}의${m[2]}` : m[1], title: m[3].trim(), body: [] };
      ensureSection().articles.push(article);
      continue;
    }
    if ((m = /^= (.*)$/.exec(line))) { push({ kind: "paragraph", text: m[1].trim(), single: true }); continue; }
    if ((m = /^@ (.*)$/.exec(line))) { push({ kind: "paragraph", text: m[1].trim() }); continue; }
    if ((m = /^    - (.*)$/.exec(line))) { push({ kind: "subitem", text: m[1].trim() }); continue; }
    if ((m = /^  - (.*)$/.exec(line))) { push({ kind: "item", text: m[1].trim() }); continue; }
    throw new Error(`알 수 없는 줄: ${line}`);
  }
  return doc;
}

function fenceToLine(f: { kind: "table" | "box" | "figure"; lines: string[] }): Line {
  if (f.kind === "box") {
    const [head, ...rest] = f.lines;
    const t = /^【(.*)】$/.exec(head ?? "");
    return { kind: "box", title: t ? t[1] : (head ?? ""), lines: t ? rest : rest.length ? rest : [] };
  }
  if (f.kind === "figure") return { kind: "box", title: "그림", lines: f.lines.map((l) => l.replace(/^설명:\s*/, "")) };
  const titleLine = f.lines.find((l) => l.startsWith("제목:"));
  const title = titleLine ? titleLine.slice(3).trim() : "(없음)";
  const rows: { header?: boolean; cells: string[] }[] = [];
  let sawSeparator = false;
  for (const l of f.lines) {
    if (!l.startsWith("|")) continue;
    if (/^\|(\s*-+\s*\|)+$/.test(l)) { sawSeparator = true; if (rows.length > 0) rows[rows.length - 1].header = true; continue; }
    rows.push({ cells: l.slice(1, l.endsWith("|") ? -1 : undefined).split("|").map((c) => c.trim()) });
  }
  void sawSeparator;
  return { kind: "table", ...(title !== "(없음)" ? { title } : {}), rows };
}

import type { InlineNode } from "../../src/domain/document/nodes";

const REF = /(보통약관 )?제(\d+)조(?:의(\d+))?\(((?:[^()]|\([^()]*\))*)\)(?: 제(\d+)항)?(?: 제(\d+)호)?/g;
const PARA = /제(\d+)항(?: 제(\d+)호)?/g;
const APX = /【별표(\d+)\(([^】]*)\)】/g;
const LAW = /(법|법률|령|규정)\s?$/;   // 「의료법 제3조」 「민법 제27조」

interface Piece { start: number; end: number; node: InlineNode }

export function inlinesFromText(text: string, env: RefEnv, newId: () => Id): InlineNode[] {
  const pieces: Piece[] = [];
  for (const m of text.matchAll(APX)) {
    const code = env.appendixByNumber.get(Number(m[1]));
    if (!code) { env.report.push(`별표 ${m[1]} 코드 없음: ${m[0]}`); continue; }
    pieces.push({ start: m.index!, end: m.index! + m[0].length, node: { id: newId(), kind: "appendixRef", appendixCode: code } });
  }
  // 조 참조 덩어리: ref (", " ref)* ((" 및 " | " 또는 ") ref)?
  for (const m of text.matchAll(REF)) {
    if (LAW.test(text.slice(0, m.index))) continue;
    if (pieces.some((p) => p.start <= m.index! && m.index! < p.end)) continue; // 이미 덩어리 안
    const scope = m[1] ? "general" : "self";
    const index = scope === "general" ? env.general : env.self;
    if (!index) { env.report.push(`보통약관 색인 없음: ${m[0]}`); continue; }
    const targets: { nodeId: Id }[] = [];
    let connector = "및";
    let cursor = m.index!;
    let rest = m[0];
    let mm: RegExpExecArray | null = m as RegExpExecArray;
    let end = m.index! + m[0].length;
    let ok = true;
    while (mm) {
      const t = resolveTarget(mm, index, env);
      if (!t) { ok = false; break; }
      targets.push({ nodeId: t });
      end = cursor + mm[0].length;
      // 다음 연결
      const tail = text.slice(end);
      const join = /^(, | 및 | 또는 )(?!보통약관 )(?=제\d+조)/.exec(tail);
      if (!join) break;
      if (join[1] !== ", ") connector = join[1].trim();
      const next = new RegExp(REF.source).exec(tail.slice(join[0].length));
      if (!next || next.index !== 0) break;
      cursor = end + join[0].length;
      mm = next;
      rest = next[0];
    }
    void rest;
    if (!ok) continue;
    pieces.push({ start: m.index!, end, node: { id: newId(), kind: "articleRef", targets, connector, scope } });
  }
  // 조 없는 항 참조 (같은 조)
  for (const m of text.matchAll(PARA)) {
    if (pieces.some((p) => p.start <= m.index! && m.index! < p.end)) continue;
    const before = text.slice(Math.max(0, m.index! - 1), m.index!);
    if (before === ")") continue; // 「제N조(…) 제M항」 은 위에서 처리
    const here = [...env.self.byNumber.values()].find((a) => a.id === env.currentArticleId);
    const pid = here?.paragraphIds[Number(m[1]) - 1];
    if (!pid) { env.report.push(`항 참조 미해소: ${m[0]} (조 ${env.currentArticleId})`); continue; }
    let target = pid;
    if (m[2]) { const iid = here?.itemIds.get(pid)?.[Number(m[2]) - 1]; if (!iid) { env.report.push(`호 참조 미해소: ${m[0]}`); continue; } target = iid; }
    pieces.push({ start: m.index!, end: m.index! + m[0].length, node: { id: newId(), kind: "articleRef", targets: [{ nodeId: target }], connector: "및", scope: "self" } });
  }
  pieces.sort((a, b) => a.start - b.start);
  const out: InlineNode[] = [];
  let pos = 0;
  for (const p of pieces) {
    if (p.start > pos) out.push({ id: newId(), kind: "text", text: text.slice(pos, p.start) });
    out.push(p.node);
    pos = p.end;
  }
  if (pos < text.length) out.push({ id: newId(), kind: "text", text: text.slice(pos) });
  return out;
}

function resolveTarget(m: RegExpExecArray, index: ArticleIndex, env: RefEnv): Id | undefined {
  const number = m[3] ? `${m[2]}의${m[3]}` : m[2];
  const a = index.byNumber.get(number);
  if (!a) { env.report.push(`조 없음: ${m[0]}`); return undefined; }
  if (a.title.replace(/\s+/g, "") !== m[4].replace(/\s+/g, "")) { env.report.push(`조 제목 불일치: ${m[0]} ↔ ${a.title}`); return undefined; }
  if (!m[5]) return a.id;
  const pid = a.paragraphIds[Number(m[5]) - 1];
  if (!pid) { env.report.push(`항 없음: ${m[0]}`); return undefined; }
  if (!m[6]) return pid;
  const iid = a.itemIds.get(pid)?.[Number(m[6]) - 1];
  if (!iid) { env.report.push(`호 없음: ${m[0]}`); return undefined; }
  return iid;
}

export function parseTableForm(fd: FormData): { title?: string; columns: TableColumn[]; rows: TableRow[] } {
  const title = str(fd, "title");
  const headerRows = Number(str(fd, "headerRows") || "0");
  const raw = String(fd.get("rows") ?? "").split(/\r?\n/).filter((l) => l.trim() !== "").map((l) => l.split("|").map((c) => c.trim()));
  const n = Math.max(1, ...raw.map((r) => r.length), str(fd, "widths") ? str(fd, "widths").split(",").length : 0);
  const widths = str(fd, "widths").split(",").map((w) => Number(w.trim()));
  const columns: TableColumn[] = Array.from({ length: n }, (_x, i) => (Number.isFinite(widths[i]) && widths[i] > 0 ? { width: widths[i] } : {}));
  const rows: TableRow[] = raw.map((cells, i) => ({ ...(i < headerRows ? { header: true } : {}), cells: [...cells, ...Array(n - cells.length).fill("")] }));
  return { ...(title ? { title } : {}), columns, rows };
}
```
